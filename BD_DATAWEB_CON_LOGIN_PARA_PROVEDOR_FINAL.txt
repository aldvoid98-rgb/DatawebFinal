
/* ============================================================
   PROYECTO DATAWEB
   SISTEMA DE GESTION DE MAQUINARIA PARA IMPRENTA
   BASE DE DATOS COMPLETA CON ACCESO PARA PROVEEDORES
   VERSION CORREGIDA: login de los 3 roles + eliminacion logica
   MYSQL 8.0.16 O SUPERIOR
   ============================================================ */

DROP DATABASE IF EXISTS DATAWEB;

CREATE DATABASE DATAWEB
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE DATAWEB;

/* ============================================================
   1. ROLES
   ============================================================ */

CREATE TABLE tb_rol (
    id_rol INT AUTO_INCREMENT PRIMARY KEY,
    nomRol VARCHAR(50) NOT NULL UNIQUE,
    descripcion VARCHAR(150),
    estado BOOLEAN NOT NULL DEFAULT TRUE
);

/* ============================================================
   2. CATEGORIAS
   ============================================================ */

CREATE TABLE tb_categoria_maquinaria (
    id_categoria_maquinaria INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL UNIQUE,
    descrip_maqui VARCHAR(200),
    estado BOOLEAN NOT NULL DEFAULT TRUE
);

CREATE TABLE tb_categoria_cliente (
    id_categoria_cliente INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL UNIQUE,
    descrip_client VARCHAR(200),
    estado BOOLEAN NOT NULL DEFAULT TRUE
);

CREATE TABLE tb_categoria_proveedor (
    id_categoria_proveedor INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL UNIQUE,
    descrip_prov VARCHAR(200),
    estado BOOLEAN NOT NULL DEFAULT TRUE
);

CREATE TABLE tb_categoria_empleado (
    id_categoria_empleado INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL UNIQUE,
    descrip_empl VARCHAR(200),
    estado BOOLEAN NOT NULL DEFAULT TRUE
);

/* ============================================================
   3. MAQUINARIA
   ============================================================ */

CREATE TABLE tb_maquinaria (
    id_maquinaria INT AUTO_INCREMENT PRIMARY KEY,
    nomMaqui VARCHAR(100) NOT NULL,
    marca VARCHAR(80) NOT NULL,
    potenciaKw DECIMAL(10,2) NOT NULL,
    precio DECIMAL(12,2) NOT NULL,
    id_categoria_maquinaria INT NOT NULL,
    estado BOOLEAN NOT NULL DEFAULT TRUE,
    situacion TINYINT NOT NULL DEFAULT 1,

    CONSTRAINT fk_maquinaria_categoria
        FOREIGN KEY (id_categoria_maquinaria)
        REFERENCES tb_categoria_maquinaria(id_categoria_maquinaria),

    CONSTRAINT chk_situacion_maquinaria
        CHECK (situacion IN (1, 2, 3))
);

/* ============================================================
   4. CLIENTES
   ============================================================ */

CREATE TABLE tb_clientes (
    id_cliente INT AUTO_INCREMENT PRIMARY KEY,
    nomClient VARCHAR(100) NOT NULL,
    apeClient VARCHAR(100) NOT NULL,
    dniClient CHAR(8) NOT NULL UNIQUE,
    telClient VARCHAR(20),
    correClient VARCHAR(120) UNIQUE,
    direClient VARCHAR(200),
    id_categoria_cliente INT NOT NULL,
    estado BOOLEAN NOT NULL DEFAULT TRUE,

    CONSTRAINT fk_cliente_categoria
        FOREIGN KEY (id_categoria_cliente)
        REFERENCES tb_categoria_cliente(id_categoria_cliente)
);

/* ============================================================
   5. PROVEEDORES
   Cada proveedor tiene el rol PROVEEDOR.
   ============================================================ */

CREATE TABLE tb_proveedores (
    id_proveedor INT AUTO_INCREMENT PRIMARY KEY,
    razon_social VARCHAR(150) NOT NULL,
    ruc CHAR(11) NOT NULL UNIQUE,
    contacto VARCHAR(100) NOT NULL,
    telProv VARCHAR(20),
    correProv VARCHAR(120) UNIQUE,
    direProv VARCHAR(200),
    id_rol INT NOT NULL,
    id_categoria_proveedor INT NOT NULL,
    estado BOOLEAN NOT NULL DEFAULT TRUE,

    CONSTRAINT fk_proveedor_rol
        FOREIGN KEY (id_rol)
        REFERENCES tb_rol(id_rol),

    CONSTRAINT fk_proveedor_categoria
        FOREIGN KEY (id_categoria_proveedor)
        REFERENCES tb_categoria_proveedor(id_categoria_proveedor)
);

/* ============================================================
   6. EMPLEADOS
   ============================================================ */

CREATE TABLE tb_empleados (
    id_empleado INT AUTO_INCREMENT PRIMARY KEY,
    nomEmpl VARCHAR(100) NOT NULL,
    apeEmpl VARCHAR(100) NOT NULL,
    dniEmpl CHAR(8) NOT NULL UNIQUE,
    cargo VARCHAR(100),
    telEmpl VARCHAR(20),
    correEmpl VARCHAR(120) UNIQUE,
    fecha_ingreso DATE,
    id_rol INT NOT NULL,
    id_categoria_empleado INT NOT NULL,
    estado BOOLEAN NOT NULL DEFAULT TRUE,

    CONSTRAINT fk_empleado_rol
        FOREIGN KEY (id_rol)
        REFERENCES tb_rol(id_rol),

    CONSTRAINT fk_empleado_categoria
        FOREIGN KEY (id_categoria_empleado)
        REFERENCES tb_categoria_empleado(id_categoria_empleado)
);

/* ============================================================
   7. CUENTAS DE ACCESO
   Una cuenta pertenece a un empleado O a un proveedor.
   Nunca a ambos al mismo tiempo.
   ============================================================ */

CREATE TABLE tb_contrasena (
    id_contrasena INT AUTO_INCREMENT PRIMARY KEY,
    id_empleado INT NULL UNIQUE,
    id_proveedor INT NULL UNIQUE,
    usuario VARCHAR(120) NOT NULL UNIQUE,
    contrasena VARCHAR(255) NOT NULL,
    fecha_creacion DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    ultimo_acceso DATETIME NULL,
    estado BOOLEAN NOT NULL DEFAULT TRUE,

    CONSTRAINT fk_contrasena_empleado
        FOREIGN KEY (id_empleado)
        REFERENCES tb_empleados(id_empleado),

    CONSTRAINT fk_contrasena_proveedor
        FOREIGN KEY (id_proveedor)
        REFERENCES tb_proveedores(id_proveedor),

    CONSTRAINT chk_contrasena_propietario
        CHECK (
            (id_empleado IS NOT NULL AND id_proveedor IS NULL)
            OR
            (id_empleado IS NULL AND id_proveedor IS NOT NULL)
        )
);

/* ============================================================
   16. VERIFICAR CUENTAS Y ROLES
   ============================================================ */

SELECT
    c.id_contrasena,
    c.usuario,
    c.contrasena,
    e.nomEmpl,
    p.razon_social,
    r.nomRol AS rol,
    c.estado AS estado_cuenta
FROM tb_contrasena c
LEFT JOIN tb_empleados e
    ON e.id_empleado = c.id_empleado
LEFT JOIN tb_proveedores p
    ON p.id_proveedor = c.id_proveedor
LEFT JOIN tb_rol r
    ON r.id_rol = COALESCE(e.id_rol, p.id_rol)
ORDER BY c.id_contrasena;

/* ============================================================
   17. ELIMINACION LOGICA
   El sistema NO borra registros (no usa DELETE). El boton
   "Eliminar" cambia la columna ESTADO de 1 (activo) a 0
   (eliminado) en tb_maquinaria, tb_clientes, tb_empleados y
   tb_proveedores. Los listados y reportes solo muestran los
   registros con estado = 1.

   Al eliminar un EMPLEADO o un PROVEEDOR, el sistema tambien
   desactiva su cuenta en tb_contrasena (estado = 0) para que
   ya no pueda iniciar sesion.

   Esta version no necesita columnas nuevas: se usa el campo
   ESTADO que ya existe en cada tabla.
   ============================================================ */

/* Ejemplos (quitar los comentarios solo si se desea probar a mano):

   -- Eliminar logicamente la maquinaria 10
   UPDATE tb_maquinaria SET estado = FALSE WHERE id_maquinaria = 10;

   -- Restaurar la maquinaria 10
   UPDATE tb_maquinaria SET estado = TRUE WHERE id_maquinaria = 10;

   -- Eliminar logicamente a un proveedor y bloquear su cuenta
   UPDATE tb_proveedores SET estado = FALSE WHERE id_proveedor = 1;
   UPDATE tb_contrasena  SET estado = FALSE WHERE id_proveedor = 1;
*/

/* ============================================================
   18. VERIFICACION: REGISTROS ACTIVOS Y ELIMINADOS
   ============================================================ */

SELECT 'tb_maquinaria' AS tabla,
       SUM(estado = 1) AS activos,
       SUM(estado = 0) AS eliminados
FROM tb_maquinaria
UNION ALL
SELECT 'tb_clientes', SUM(estado = 1), SUM(estado = 0) FROM tb_clientes
UNION ALL
SELECT 'tb_empleados', SUM(estado = 1), SUM(estado = 0) FROM tb_empleados
UNION ALL
SELECT 'tb_proveedores', SUM(estado = 1), SUM(estado = 0) FROM tb_proveedores;
